//   - Для данных, которые получаем во время работы программы

КОГДА ИСПОЛЬЗОВАТЬ:
.toString()     → Когда нужно ПРЕОБРАЗОВАТЬ в строку
as String       → Когда уверены, что объект УЖЕ строка
as? String      → Когда не уверены в типе
is String       → Для безопасной проверки типа
int.tryParse()  → Для безопасного преобразования строк в числа

for (var fruit in fruits) {
  print(fruit); // ✅ Просто и читаемо
}

// 2. Перебор Set
var uniqueNumbers = {1, 2, 3, 4};
for (var number in uniqueNumbers) {
  print(number);
}

// 3. Перебор ключей Map
var person = {'name': 'John', 'age': 30};
for (var key in person.keys) {
  print('$key: ${person[key]}');
}

// 4. С where (фильтрация)
for (var number in [1, 2, 3, 4, 5].where((n) => n % 2 == 0)) {
  print('Even: $number'); // 2, 4
}

// ============================ WHILE ============================
// ✅ Когда количество итераций неизвестно
// ✅ Условие проверяется ДО выполнения цикла
// ✅ Может не выполниться ни разу

// 🎯 ПРИМЕРЫ WHILE:

// 1. Чтение до конца файла/потока
var hasData = true;
while (hasData) {
  // hasData = readNextData(); // ✅ Пока есть данные
  print('Processing data...');
  hasData = false; // Для примера
}

// 2. Ожидание условия
var isLoading = true;
while (isLoading) {
  print('Waiting...');
  // isLoading = checkStatus(); // ✅ Пока грузится
  isLoading = false;
}

// 3. Обработка до выполнения условия
var number = 10;
while (number > 0) {
  print(number);
  number--; // ✅ Уменьшаем до нуля
}

// ============================ DO-WHILE ============================
// ✅ Когда нужно выполнить минимум ОДИН раз
// ✅ Условие проверяется ПОСЛЕ выполнения цикла
// ✅ Полезно для меню, валидации ввода

// 🎯 ПРИМЕРЫ DO-WHILE:

// 1. Ввод пользователя (минимум 1 раз)
var input;
do {
  print('Enter "yes" to continue:');
  input = 'yes'; // Имитация ввода
} while (input != 'yes');

// 2. Генерация до условия
var randomNumber;
do {
  randomNumber = 5; // Имитация random
  print('Generated: $randomNumber');
} while (randomNumber < 3); // ✅ Хотя бы один раз выполнится

// 3. Повторение действия
var attempt = 0;
do {
  attempt++;
  print('Attempt: $attempt');
  // var success = tryOperation(); // ✅ Пока не получится
} while (attempt < 3);

//   - Известное количество итераций
//   - Нужен индекс
//   - Сложные условия инкремента
//   - Обратный перебор

// 🔄 FOR-IN:
//   - Простой перебор коллекций
//   - Не нужен индекс
//   - Лучшая читаемость
//   - Для Iterable объектов

//   - Неизвестное количество итераций
//   - Условие может не выполниться
//   - Ожидание внешних условий
//   - Проверка условия ДО выполнения

// 🔄 DO-WHILE:
//   - Минимум одна итерация гарантирована
//   - Валидация пользовательского ввода
//   - Проверка условия ПОСЛЕ выполнения

// ===================== ПРАКТИЧЕСКИЕ ПРАВИЛА =====================

// ✅ Правило 1: Для коллекций с нужным индексом -> FOR
for (var i = 0; i < list.length; i++) {
  if (i % 2 == 0) print(list[i]); // ✅ Четные индексы
}

// ✅ Правило 2: Для простого обхода элементов -> FOR-IN  
for (var item in list) {
  print(item); // ✅ Просто и понятно
}

// ✅ Правило 3: Когда неизвестно сколько раз -> WHILE
while (shouldContinue()) { // ✅ Пока условие истинно
  // обработка
}

// ✅ Правило 4: Когда минимум один раз -> DO-WHILE
do {
  var valid = validateInput(); // ✅ Хотя бы одна проверка
} while (!valid);

----------------------------Late-----------------------

КОГДА ИСПОЛЬЗОВАТЬ late:
✅ Гарантированная инициализация в конструкторе
✅ Инициализация в initState()/init()
✅ Ленивые вычисления (тяжелые операции)
✅ DI/Service Location паттерны

КОГДА НЕ ИСПОЛЬЗОВАТЬ:
❌ Значение может быть null
❌ Нет гарантии инициализации
❌ Сложные/асинхронные инициализации

ВИБОР МЕЖДУ:
late var           // Изменяемое, отложенная инициализация
late final         // Однократная инициализация
late final = ...   // Ленивое вычисление

// 1. Буду ли я гарантированно инициализировать это поле ДО использования?
// ДА → late

// 2. Может ли это поле остаться null в некоторых сценариях?
// НЕТ → late final

// 3. Нужно ли отложить тяжелые вычисления до первого использования?
// ДА → late final = вычисление

// 4. Использую ли я Flutter и инициализацию в initState?
// ДА → late
